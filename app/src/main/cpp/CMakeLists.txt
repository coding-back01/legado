cmake_minimum_required(VERSION 3.22.1)
project(legado_dictionary C)
set(CMAKE_C_STANDARD 99)
set(MOBI_SOURCES buffer compression debug index memory meta parse_rawml read structure util)
set(SOURCES dictionary_mobi.c)
foreach(SOURCE IN LISTS MOBI_SOURCES)
    list(APPEND SOURCES "libmobi/${SOURCE}.c")
endforeach()
add_library(legado_dictionary SHARED ${SOURCES})
target_include_directories(legado_dictionary PRIVATE libmobi)
target_compile_definitions(legado_dictionary PRIVATE HAVE_STRDUP HAVE_UNISTD_H MOBI_INLINE=inline RAWTEXT_SIZEMAX=536870912)
target_compile_options(legado_dictionary PRIVATE -fvisibility=hidden)
target_link_options(legado_dictionary PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(legado_dictionary z)
